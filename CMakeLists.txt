cmake_minimum_required(VERSION 3.24)
set(GHE_VERSION "0.2.5" CACHE STRING "Application version")
if(DEFINED ENV{GITHUB_REF_TYPE} AND "$ENV{GITHUB_REF_TYPE}" STREQUAL "tag" AND "$ENV{GITHUB_REF_NAME}" MATCHES "^v([0-9]+\.[0-9]+\.[0-9]+)$")
    set(GHE_VERSION "${CMAKE_MATCH_1}")
endif()
project(GitHubEnhanced VERSION ${GHE_VERSION} LANGUAGES CXX)
configure_file(src/appversion.h.in "${CMAKE_CURRENT_BINARY_DIR}/generated/appversion.h" @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
set(INTEGRATION_SOURCES src/mergedialog.cpp src/mergedialog.h src/appupdater.cpp src/appupdater.h src/updatedialog.cpp src/updatedialog.h src/custompanel.cpp src/custompanel.h src/diagnostics.cpp src/diagnostics.h src/repositorycreateform.cpp src/repositorycreateform.h src/releasesdialog.cpp src/releasesdialog.h
    src/pushpreflight.cpp src/pushpreflight.h
    src/repositorysync.cpp src/repositorysync.h src/capsulescrollbar.cpp src/capsulescrollbar.h
    src/operationdialog.cpp src/operationdialog.h src/repositoryindex.cpp src/repositoryindex.h
    src/toolruntime.cpp src/toolruntime.h src/codexclient.cpp src/codexclient.h
    src/githubclient.cpp src/githubclient.h src/accountdialog.cpp src/accountdialog.h
    src/repobrowser.cpp src/repobrowser.h src/reposettings.cpp src/reposettings.h
    src/layoutcanvas.cpp src/layoutcanvas.h)
qt_add_executable(github-enhanced WIN32 MACOSX_BUNDLE
    src/main.cpp src/mainwindow.cpp src/mainwindow.h
    src/commandrunner.cpp src/commandrunner.h
    src/gitstatus.cpp src/gitstatus.h
    src/theme.cpp src/theme.h src/settingsdialog.cpp src/settingsdialog.h src/layoutdialog.cpp src/layoutdialog.h
    src/visuals.cpp src/visuals.h src/glassstyle.cpp src/glassstyle.h ${INTEGRATION_SOURCES})
target_link_libraries(github-enhanced PRIVATE Qt6::Widgets Qt6::Network)
target_include_directories(github-enhanced PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
qt_add_resources(github-enhanced themes PREFIX "/" FILES
    themes/midnight.json themes/daylight.json themes/forest.json themes/obsidian.json themes/graphite.json themes/paper.json themes/ocean.json favicon.svg packaging/icons/github-enhanced.png)
set_target_properties(github-enhanced PROPERTIES
    MACOSX_BUNDLE_BUNDLE_NAME "GitHub Enhanced"
    MACOSX_BUNDLE_GUI_IDENTIFIER "io.githubenhanced.desktop"
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
if(WIN32)
    enable_language(RC)
    configure_file(packaging/windows/icon.rc.in "${CMAKE_CURRENT_BINARY_DIR}/icon.rc" @ONLY)
    target_sources(github-enhanced PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/icon.rc")
elseif(APPLE)
    target_sources(github-enhanced PRIVATE packaging/icons/github-enhanced.icns)
    set_source_files_properties(packaging/icons/github-enhanced.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    set_target_properties(github-enhanced PROPERTIES MACOSX_BUNDLE_ICON_FILE github-enhanced.icns)
endif()
if(MSVC)
    target_compile_options(github-enhanced PRIVATE /W4 /utf-8)
else()
    target_compile_options(github-enhanced PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(GNUInstallDirs)
install(TARGETS github-enhanced BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(UNIX AND NOT APPLE)
    install(FILES packaging/linux/io.githubenhanced.desktop.desktop DESTINATION share/applications RENAME io.githubenhanced.GitHubEnhanced.desktop)
    install(FILES favicon.svg DESTINATION share/icons/hicolor/scalable/apps RENAME io.githubenhanced.GitHubEnhanced.svg)
    install(FILES packaging/icons/github-enhanced.png DESTINATION share/icons/hicolor/1024x1024/apps RENAME io.githubenhanced.GitHubEnhanced.png)
endif()
if(WIN32 OR APPLE)
    qt_generate_deploy_app_script(TARGET github-enhanced OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
    install(SCRIPT ${deploy_script})
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(core-tests tests/core_tests.cpp src/commandrunner.cpp src/commandrunner.h src/gitstatus.cpp src/theme.cpp src/visuals.cpp src/visuals.h src/glassstyle.cpp src/glassstyle.h src/capsulescrollbar.cpp src/capsulescrollbar.h)
    target_include_directories(core-tests PRIVATE src)
    target_link_libraries(core-tests PRIVATE Qt6::Widgets)
    add_test(NAME core COMMAND core-tests)
    set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
    qt_add_executable(ui-tests tests/ui_tests.cpp src/mainwindow.cpp src/mainwindow.h
        src/settingsdialog.cpp src/settingsdialog.h src/layoutdialog.cpp src/layoutdialog.h src/theme.cpp
        src/gitstatus.cpp src/commandrunner.cpp src/commandrunner.h src/visuals.cpp src/visuals.h src/glassstyle.cpp src/glassstyle.h ${INTEGRATION_SOURCES})
    target_include_directories(ui-tests PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_link_libraries(ui-tests PRIVATE Qt6::Widgets Qt6::Network Qt6::Test)
    qt_add_resources(ui-tests themes PREFIX "/" FILES themes/midnight.json themes/daylight.json themes/forest.json themes/obsidian.json themes/graphite.json themes/paper.json themes/ocean.json favicon.svg packaging/icons/github-enhanced.png)
    if(MSVC)
        target_compile_options(core-tests PRIVATE /utf-8)
        target_compile_options(ui-tests PRIVATE /utf-8)
    endif()
    add_test(NAME ui-workflow COMMAND ui-tests -o -,txt -o qt-ui-results.txt,txt)
    set_tests_properties(ui-workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
    qt_add_executable(fake-services tests/fake_services.cpp)
    target_link_libraries(fake-services PRIVATE Qt6::Core)
    qt_add_executable(integration-tests tests/integration_tests.cpp ${INTEGRATION_SOURCES}
        src/commandrunner.cpp src/commandrunner.h src/gitstatus.cpp src/visuals.cpp src/visuals.h src/glassstyle.cpp src/glassstyle.h src/theme.cpp)
    target_include_directories(integration-tests PRIVATE src)
    target_link_libraries(integration-tests PRIVATE Qt6::Widgets Qt6::Network Qt6::Test)
    add_dependencies(integration-tests fake-services)
    add_dependencies(ui-tests fake-services)
    if(MSVC)
        target_compile_options(fake-services PRIVATE /utf-8)
        target_compile_options(integration-tests PRIVATE /utf-8)
    endif()
    # Qt marks executables as GUI apps on Windows; tests need console output.
    foreach(test_target core-tests ui-tests integration-tests fake-services)
        set_target_properties(${test_target} PROPERTIES WIN32_EXECUTABLE FALSE)
    endforeach()
    add_test(NAME accounts-integration COMMAND integration-tests -o -,txt -o qt-integration-results.txt,txt)
    set_tests_properties(accounts-integration PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
endif()
if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 20)
    foreach(app_target github-enhanced ui-tests core-tests integration-tests)
        if(TARGET ${app_target})
            set_target_properties(${app_target} PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED ON)
            target_sources(${app_target} PRIVATE src/nativeglass_mac.mm)
            target_compile_options(${app_target} PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
            target_link_libraries(${app_target} PRIVATE "-framework AppKit" "-framework QuartzCore")
        endif()
    endforeach()
endif()
set(CPACK_PACKAGE_NAME "github-enhanced")
set(CPACK_PACKAGE_VENDOR "GitHub Enhanced contributors")
set(CPACK_PACKAGE_CONTACT "GitHub Enhanced contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Customizable native Git desktop client")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
# WiX accepts .txt/.rtf licenses, but rejects an extensionless LICENSE.
configure_file(LICENSE "${CMAKE_CURRENT_BINARY_DIR}/LICENSE.txt" COPYONLY)
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_BINARY_DIR}/LICENSE.txt")
set(CPACK_PACKAGE_INSTALL_DIRECTORY "GitHub Enhanced")
set(CPACK_PACKAGE_EXECUTABLES "github-enhanced;GitHub Enhanced")
set(CPACK_SOURCE_GENERATOR "TGZ")
set(CPACK_SOURCE_PACKAGE_FILE_NAME "github-enhanced-${PROJECT_VERSION}")
set(CPACK_SOURCE_IGNORE_FILES "/build[^/]*/;/dist/;/Virtual-Devices/;/.git/;/.tools/;\\.AppImage$;\\.user$")
if(WIN32)
    set(CPACK_GENERATOR "NSIS;WIX")
    set(CPACK_WIX_UPGRADE_GUID "19883D28-94B2-46D3-A178-861535AD934B")
    set(CPACK_WIX_VERSION 4)
    set(CPACK_NSIS_MODIFY_PATH OFF)
    set(CPACK_NSIS_MUI_ICON "${CMAKE_SOURCE_DIR}/packaging/icons/github-enhanced.ico")
    set(CPACK_WIX_PRODUCT_ICON "${CMAKE_SOURCE_DIR}/packaging/icons/github-enhanced.ico")
elseif(APPLE)
    if("arm64" IN_LIST CMAKE_OSX_ARCHITECTURES AND "x86_64" IN_LIST CMAKE_OSX_ARCHITECTURES)
        set(CPACK_PACKAGE_FILE_NAME "github-enhanced-${PROJECT_VERSION}-macos-universal")
    else()
        set(CPACK_PACKAGE_FILE_NAME "github-enhanced-${PROJECT_VERSION}-macos-${CMAKE_SYSTEM_PROCESSOR}")
    endif()
    set(CPACK_GENERATOR "DragNDrop")
else()
    set(CPACK_GENERATOR "DEB;RPM;TGZ")
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "git")
    set(CPACK_RPM_PACKAGE_REQUIRES "git")
endif()
include(CPack)
