cmake_minimum_required(VERSION 3.24)
project(GitHubEnhanced VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
set(INTEGRATION_SOURCES src/repositorycreateform.cpp src/repositorycreateform.h src/releasesdialog.cpp src/releasesdialog.h
    src/pushpreflight.cpp src/pushpreflight.h
    src/repositorysync.cpp src/repositorysync.h src/capsulescrollbar.cpp src/capsulescrollbar.h
    src/operationdialog.cpp src/operationdialog.h src/repositoryindex.cpp src/repositoryindex.h
    src/toolruntime.cpp src/toolruntime.h src/codexclient.cpp src/codexclient.h
    src/githubclient.cpp src/githubclient.h src/accountdialog.cpp src/accountdialog.h
    src/repobrowser.cpp src/repobrowser.h src/reposettings.cpp src/reposettings.h
    src/layoutcanvas.cpp src/layoutcanvas.h)
qt_add_executable(github-enhanced WIN32 MACOSX_BUNDLE
    src/main.cpp src/mainwindow.cpp src/mainwindow.h
    src/commandrunner.cpp src/commandrunner.h
    src/gitstatus.cpp src/gitstatus.h
    src/theme.cpp src/theme.h src/settingsdialog.cpp src/settingsdialog.h src/layoutdialog.cpp src/layoutdialog.h
    src/visuals.cpp src/visuals.h ${INTEGRATION_SOURCES})
target_link_libraries(github-enhanced PRIVATE Qt6::Widgets Qt6::Network)
qt_add_resources(github-enhanced themes PREFIX "/" FILES
    themes/midnight.json themes/daylight.json themes/forest.json themes/obsidian.json)
set_target_properties(github-enhanced PROPERTIES
    MACOSX_BUNDLE_BUNDLE_NAME "GitHub Enhanced"
    MACOSX_BUNDLE_GUI_IDENTIFIER "io.githubenhanced.desktop"
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
if(MSVC)
    target_compile_options(github-enhanced PRIVATE /W4 /utf-8)
else()
    target_compile_options(github-enhanced PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(GNUInstallDirs)
install(TARGETS github-enhanced BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(UNIX AND NOT APPLE)
    install(FILES packaging/linux/io.githubenhanced.desktop.desktop DESTINATION share/applications)
    install(FILES packaging/linux/io.githubenhanced.desktop.svg DESTINATION share/icons/hicolor/scalable/apps)
endif()
if(WIN32 OR APPLE)
    qt_generate_deploy_app_script(TARGET github-enhanced OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
    install(SCRIPT ${deploy_script})
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(core-tests tests/core_tests.cpp src/commandrunner.cpp src/commandrunner.h src/gitstatus.cpp src/theme.cpp src/visuals.cpp src/visuals.h src/capsulescrollbar.cpp src/capsulescrollbar.h)
    target_include_directories(core-tests PRIVATE src)
    target_link_libraries(core-tests PRIVATE Qt6::Widgets)
    add_test(NAME core COMMAND core-tests)
    set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    qt_add_executable(ui-tests tests/ui_tests.cpp src/mainwindow.cpp src/mainwindow.h
        src/settingsdialog.cpp src/settingsdialog.h src/layoutdialog.cpp src/layoutdialog.h src/theme.cpp
        src/gitstatus.cpp src/commandrunner.cpp src/commandrunner.h src/visuals.cpp src/visuals.h ${INTEGRATION_SOURCES})
    target_include_directories(ui-tests PRIVATE src)
    target_link_libraries(ui-tests PRIVATE Qt6::Widgets Qt6::Network Qt6::Test)
    qt_add_resources(ui-tests themes PREFIX "/" FILES themes/midnight.json themes/daylight.json themes/forest.json themes/obsidian.json)
    if(MSVC)
        target_compile_options(core-tests PRIVATE /utf-8)
        target_compile_options(ui-tests PRIVATE /utf-8)
    endif()
    add_test(NAME ui-workflow COMMAND ui-tests)
    set_tests_properties(ui-workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
    qt_add_executable(fake-services tests/fake_services.cpp)
    target_link_libraries(fake-services PRIVATE Qt6::Core)
    qt_add_executable(integration-tests tests/integration_tests.cpp ${INTEGRATION_SOURCES}
        src/commandrunner.cpp src/commandrunner.h src/gitstatus.cpp src/visuals.cpp src/visuals.h src/theme.cpp)
    target_include_directories(integration-tests PRIVATE src)
    target_link_libraries(integration-tests PRIVATE Qt6::Widgets Qt6::Network Qt6::Test)
    add_dependencies(integration-tests fake-services)
    add_dependencies(ui-tests fake-services)
    if(MSVC)
        target_compile_options(fake-services PRIVATE /utf-8)
        target_compile_options(integration-tests PRIVATE /utf-8)
    endif()
    add_test(NAME accounts-integration COMMAND integration-tests)
    set_tests_properties(accounts-integration PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
endif()
if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 20)
    foreach(app_target github-enhanced ui-tests core-tests integration-tests)
        if(TARGET ${app_target})
            set_target_properties(${app_target} PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED ON)
            target_sources(${app_target} PRIVATE src/nativeglass_mac.mm)
            target_compile_options(${app_target} PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
            target_link_libraries(${app_target} PRIVATE "-framework AppKit" "-framework QuartzCore")
        endif()
    endforeach()
endif()
set(CPACK_PACKAGE_NAME "github-enhanced")
set(CPACK_PACKAGE_VENDOR "GitHub Enhanced contributors")
set(CPACK_PACKAGE_CONTACT "GitHub Enhanced contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Customizable native Git desktop client")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_SOURCE_DIR}/LICENSE")
set(CPACK_PACKAGE_INSTALL_DIRECTORY "GitHub Enhanced")
set(CPACK_PACKAGE_EXECUTABLES "github-enhanced;GitHub Enhanced")
set(CPACK_SOURCE_GENERATOR "TGZ")
set(CPACK_SOURCE_PACKAGE_FILE_NAME "github-enhanced-${PROJECT_VERSION}")
set(CPACK_SOURCE_IGNORE_FILES "/build[^/]*/;/dist/;/Virtual-Devices/;/.git/;/.tools/;\\.AppImage$;\\.user$")
if(WIN32)
    set(CPACK_GENERATOR "NSIS;WIX")
    set(CPACK_WIX_UPGRADE_GUID "19883D28-94B2-46D3-A178-861535AD934B")
    set(CPACK_WIX_VERSION 4)
    set(CPACK_NSIS_MODIFY_PATH OFF)
elseif(APPLE)
    set(CPACK_GENERATOR "DragNDrop")
else()
    set(CPACK_GENERATOR "DEB;RPM;TGZ")
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "git")
    set(CPACK_RPM_PACKAGE_REQUIRES "git")
endif()
include(CPack)
